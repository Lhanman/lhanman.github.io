表结构

users — 用户表

users (
    id           BIGSERIAL PRIMARY KEY,
    username     VARCHAR(100) UNIQUE,
    email        VARCHAR(255) UNIQUE,
    password_hash VARCHAR(255),   -- BCrypt 加密
    role         VARCHAR(20),     -- USER / ADMIN
    created_at   TIMESTAMPTZ
)

posts — 文章表

posts (
    id             BIGSERIAL PRIMARY KEY,
    title          VARCHAR(500),
    slug           VARCHAR(500) UNIQUE,   -- URL 友好标识
    summary        TEXT,
    content        TEXT,
    cover_image_url TEXT,
    published      BOOLEAN,               -- 草稿/发布状态
    author_id      BIGINT → users(id),
    created_at     TIMESTAMPTZ,
    updated_at     TIMESTAMPTZ,
    search_vector  tsvector               -- 全文搜索向量（自动生成）
)

tags — 标签表

tags (
    id   BIGSERIAL PRIMARY KEY,
    name VARCHAR(100) UNIQUE,
    slug VARCHAR(100) UNIQUE
)

post_tags — 文章标签关联表（多对多）

post_tags (
    post_id BIGINT → posts(id) ON DELETE CASCADE,
    tag_id  BIGINT → tags(id) ON DELETE CASCADE,
    PRIMARY KEY (post_id, tag_id)
)

comments — 评论表

comments (
    id         BIGSERIAL PRIMARY KEY,
    post_id    BIGINT → posts(id) ON DELETE CASCADE,
    user_id    BIGINT → users(id),
    content    TEXT,
    created_at TIMESTAMPTZ,
    is_deleted BOOLEAN   -- 软删除
)

索引

CREATE INDEX posts_search_idx ON posts USING GIN(search_vector);

数据库迁移

使用 Flyway 管理，迁移文件位于 backend/src/main/resources/db/migration/

命名格式：V{version}__{description}.sql，例如 V2__add_user_avatar.sql

重启后端后 Flyway 自动执行迁移。

注意事项

  • 时间戳字段使用 timestamp 而非 timestampWithTimeZone
  • 代码中使用 Clock.System.now() 赋值
